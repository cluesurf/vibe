\subsection{Greek}

\begin{longtable}{L{0.30\linewidth} L{0.62\linewidth}}
\toprule
\textbf{Term} & \textbf{Meaning} \\
\midrule
\endhead
Chaos & The yawning gap, the void that came first of all \\
Gaia & The broad earth, the firm seat of all things \\
Ouranos & The starry sky, born of Gaia, the first father \\
Tartaros & The misty pit deep below the earth \\
Eros & Desire, the force of joining that lets the powers breed \\
Nyx & Night, born of Chaos, mother of dark powers \\
Titans & The elder gods, children of Gaia and Ouranos \\
Kronos & The youngest Titan who unseated his father and then ruled \\
Rhea & Sister and wife of Kronos, mother of the Olympians \\
Zeus & The youngest son who overthrew Kronos and became king of the sky \\
Poseidon & Brother of Zeus, lord of the sea \\
Hades & Brother of Zeus, lord of the dead below \\
Titanomachy & The ten-year war of the young gods against the Titans \\
Olympus & The mountain home of the twelve great gods \\
Moirai & The three Fates who spin, measure, and cut each life \\
Hades (place) & The underworld, the realm of the dead \\
Eleusinian Mysteries & The secret rites of Demeter and Persephone \\
\bottomrule
\end{longtable}

\subsection{Egyptian}

\begin{longtable}{L{0.30\linewidth} L{0.62\linewidth}}
\toprule
\textbf{Term} & \textbf{Meaning} \\
\midrule
\endhead
Nun & The primordial dark waters before all things \\
Benben & The first mound of dry land risen from Nun \\
Atum & The self-created creator who arises on the mound \\
Shu & The principle of dry air, first of Atum's pair \\
Tefnut & The principle of moisture, paired with Shu \\
Geb & The earth, son of Shu and Tefnut \\
Nut & The sky, lifted off the earth by Shu \\
Ennead & The family of nine gods of Heliopolis \\
Osiris & The just king, later lord of the dead \\
Isis & The great magician, wife of Osiris \\
Set & The force of disorder and the desert \\
Ptah & The Memphite creator who makes by heart and tongue \\
Maat & The principle of truth, justice, and cosmic order \\
Isfet & Chaos, disorder, and falsehood, the counter-force \\
Ra & The sun god who sails the sky and the underworld \\
Apep & The serpent of chaos that attacks the sun each night \\
Duat & The underworld through which the sun and the dead pass \\
\bottomrule
\end{longtable}

\subsection{Chinese}

\begin{longtable}{L{0.30\linewidth} L{0.62\linewidth}}

\midrule
\endhead

Shennong & Culture-bringer of farming and herbal medicine \\

Huainanzi & A Han-era text gathering the Pangu and cosmic myths \\
Shanhaijing & The Classic of Mountains and Seas, a source of the myths \\
Eight trigrams & The set of symbols attributed to Fuxi \\
Tao & The unnameable source prior to all distinction \\
Ten thousand things & The full multiplicity of the manifest world \\
\bottomrule
\end{longtable}
